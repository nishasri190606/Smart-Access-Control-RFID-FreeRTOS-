#include "rc522.h"

static SPI_HandleTypeDef *rc522_spi;

#define RC522_CS_PORT GPIOA
#define RC522_CS_PIN  GPIO_PIN_4

#define RC522_RST_PORT GPIOB
#define RC522_RST_PIN  GPIO_PIN_0

#define PCD_IDLE        0x26
#define PICC_REQIDL     0x26
#define PICC_ANTICOLL   0x93

static void RC522_CS_Low(void)
{
    HAL_GPIO_WritePin(RC522_CS_PORT, RC522_CS_PIN, GPIO_PIN_RESET);
}

static void RC522_CS_High(void)
{
    HAL_GPIO_WritePin(RC522_CS_PORT, RC522_CS_PIN, GPIO_PIN_SET);
}

static uint8_t RC522_SPI_Transfer(uint8_t data)
{
    uint8_t rx;

    HAL_SPI_TransmitReceive(rc522_spi, &data, &rx, 1, HAL_MAX_DELAY);

    return rx;
}

static void RC522_WriteRegister(uint8_t address, uint8_t value)
{
    RC522_CS_Low();

    RC522_SPI_Transfer((address << 1) & 0x7E);
    RC522_SPI_Transfer(value);

    RC522_CS_High();
}

static uint8_t RC522_ReadRegister(uint8_t address)
{
    uint8_t value;

    RC522_CS_Low();

    RC522_SPI_Transfer(((address << 1) & 0x7E) | 0x80);
    value = RC522_SPI_Transfer(0x00);

    RC522_CS_High();

    return value;
}

void RC522_Init(SPI_HandleTypeDef *hspi)
{
    rc522_spi = hspi;

    HAL_GPIO_WritePin(RC522_RST_PORT, RC522_RST_PIN, GPIO_PIN_RESET);
    HAL_Delay(10);

    HAL_GPIO_WritePin(RC522_RST_PORT, RC522_RST_PIN, GPIO_PIN_SET);
    HAL_Delay(50);

    RC522_WriteRegister(0x01, PCD_IDLE);
}
